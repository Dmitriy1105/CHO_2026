# ============================================================
# 3. FILE CONTENT SEARCH
# ============================================================

def file_contains_token(
    path,
    token,
    chunk_size=1024 * 1024
):
    """
    Поиск текстового маркера во всём файле
    с последовательным чтением блоков.

    Учитывается возможное пересечение маркера
    между соседними блоками.
    """

    token = token.encode("ascii")

    overlap_size = max(
        len(token) - 1,
        0
    )

    overlap = b""

    with open(path, "rb") as file:

        while True:

            chunk = file.read(
                chunk_size
            )

            if not chunk:
                break

            data = overlap + chunk

            if token in data:
                return True

            overlap = data[-overlap_size:] if overlap_size else b""

    return False


# ============================================================
# 4. CIF FILE VALIDATION
# ============================================================

def validate_cif_file(path):
    """
    Минимальная структурная проверка mmCIF.

    Проверяется:
    - существование файла;
    - минимальный размер;
    - data-блок в начале файла;
    - наличие atom_site во всём файле.

    Важно:
    atom_site может располагаться значительно
    дальше первых 100 KB.
    """

    path = Path(path)

    if not path.exists():
        return False

    try:

        if path.stat().st_size < 100:
            return False

        # Проверяем начало файла.
        with open(path, "rb") as file:

            header = file.read(4096)

        if not header.strip():
            return False

        if b"data_" not in header:
            return False

        # Проверяем atom_site по всему файлу.
        if not file_contains_token(
            path,
            "_atom_site."
        ):
            return False

        return True

    except OSError:
        return False
